# Deployment resource tags. Terraform reads TF_VAR_* values after .env is sourced.
# $USER is evaluated when each colleague sources this file, so do not replace it
# with a literal personal username in this repository.
export TF_VAR_created_by="${USER:-${USERNAME:-unknown}}"

# Generate once in the current shell when this file is sourced. A pre-existing
# TF_VAR_build_ver wins, which lets CI or a wrapper pin a specific identifier.
# Keep sourcing the same shell for repeated plan/apply iterations; regenerate it
# only when intentionally starting a new deployment version.
export TF_VAR_build_ver="${TF_VAR_build_ver:-$(date -u '+%Y-%m-%dT%H:%M:%SZ')}"
